cmake_minimum_required(VERSION 3.22)
project(vox_speech LANGUAGES C CXX)
include(FetchContent)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(whisper URL https://github.com/ggml-org/whisper.cpp/archive/refs/tags/v1.9.5.tar.gz)
FetchContent_MakeAvailable(whisper)
add_library(vox_speech SHARED speech_jni.cpp)
target_compile_features(vox_speech PRIVATE cxx_std_17)
target_link_libraries(vox_speech PRIVATE whisper android log)
